import logging
import os
import sys

from telegram import Update
from telegram.ext import (
    Application,
    CommandHandler,
    MessageHandler,
    CallbackQueryHandler,
    filters,
    ContextTypes,
    ConversationHandler,
)
from dotenv import load_dotenv

from handlers.start import start_command, button_handler
from handlers.excel_handler import (
    receive_excel,
    analyze_data,
    ASK_ANALYSIS_TYPE,
    cancel_handler,
)
from handlers.report_handler import (
    balance_sheet_command,
    income_statement_command,
    cash_flow_command,
    quick_analysis_command,
)

logging.basicConfig(
    format="%(asctime)s - %(name)s - %(levelname)s - %(message)s",
    level=logging.INFO,
    handlers=[
        logging.FileHandler("bot.log", encoding="utf-8"),
        logging.StreamHandler(sys.stdout),
    ],
)
logger = logging.getLogger(__name__)

load_dotenv()

BOT_TOKEN = os.getenv("BOT_TOKEN")
ADMIN_ID = os.getenv("ADMIN_ID")

if not BOT_TOKEN:
    logger.error("BOT_TOKEN topilmadi! .env faylni tekshiring.")
    sys.exit(1)


def main() -> None:
    logger.info("Bot ishga tushirilmoqda...")
    application = Application.builder().token(BOT_TOKEN).build()

    excel_conv_handler = ConversationHandler(
        entry_points=[
            MessageHandler(
                filters.Document.FileExtension("xlsx") | filters.Document.FileExtension("xls"),
                receive_excel,
            )
        ],
        states={
            ASK_ANALYSIS_TYPE: [
                MessageHandler(filters.TEXT & ~filters.COMMAND, analyze_data),
            ],
        },
        fallbacks=[
            CommandHandler("cancel", cancel_handler),
            MessageHandler(filters.COMMAND, cancel_handler),
        ],
    )

    application.add_handler(CommandHandler("start", start_command))
    application.add_handler(CommandHandler("balans", balance_sheet_command))
    application.add_handler(CommandHandler("daromad", income_statement_command))
    application.add_handler(CommandHandler("naqd", cash_flow_command))
    application.add_handler(CommandHandler("tezkor", quick_analysis_command))
    application.add_handler(CommandHandler("yordam", start_command))
    application.add_handler(CallbackQueryHandler(button_handler))
    application.add_handler(excel_conv_handler)
    application.add_error_handler(error_handler)

    logger.info("Bot muvaffaqiyatli ishga tushdi!")
    application.run_polling(allowed_updates=Update.ALL_TYPES)


async def error_handler(update: Update, context: ContextTypes.DEFAULT_TYPE):
    logger.error(f"Xatolik: {context.error}", exc_info=True)
    if update and update.effective_message:
        await update.effective_message.reply_text(
            "❌ <b>Xatolik yuz berdi!</b>\n\nIltimos, qaytadan urinib ko'ring.",
            parse_mode="HTML",
        )


if __name__ == "__main__":
    main()

import os
import tempfile
from telegram import Update, ReplyKeyboardMarkup, ReplyKeyboardRemove
from telegram.ext import ContextTypes, ConversationHandler

from services.excel_parser import ExcelParser
from services.report_generator import ReportGenerator

ASK_ANALYSIS_TYPE = 1

ANALYSIS_TYPES = [
    ["📊 Balans tahlili"],
    ["📈 Daromadlar/xarajatlar"],
    ["🔍 Xatolarni tekshirish"],
    ["📑 To'liq hisobot"],
    ["❌ Bekor qilish"],
]


async def receive_excel(update: Update, context: ContextTypes.DEFAULT_TYPE):
    document = update.message.document

    if document.file_size > 10 * 1024 * 1024:
        await update.message.reply_text("❌ Fayl hajmi 10 MB dan oshmasligi kerak!")
        return ConversationHandler.END

    if not document.file_name.endswith((".xlsx", ".xls")):
        await update.message.reply_text("❌ Faqat .xlsx yoki .xls formatidagi fayllarni qabul qilaman!")
        return ConversationHandler.END

    processing_msg = await update.message.reply_text(
        "⏳ <b>Fayl yuklanmoqda...</b>",
        parse_mode="HTML",
    )

    try:
        file = await context.bot.get_file(document.file_id)
        temp_dir = tempfile.gettempdir()
        file_path = os.path.join(temp_dir, f"accounting_{update.effective_user.id}_{document.file_name}")

        await file.download_to_drive(file_path)

        parser = ExcelParser(file_path)

        if not parser.load_file():
            await processing_msg.edit_text(
                "❌ <b>Faylni o'qishda xatolik yuz berdi!</b>\n\n"
                "Fayl to'g'ri Excel formatida ekanligini tekshiring."
            )
            return ConversationHandler.END

        is_valid, errors = parser.validate_structure()

        if not is_valid:
            error_text = "⚠️ <b>Fayl tuzilmasida xatolar:</b>\n\n"
            for i, error in enumerate(errors, 1):
                error_text += f"{i}. {error}\n"
            error_text += "\n📎 Iltimos, faylni to'g'ri formatda yuboring."

            await processing_msg.edit_text(error_text, parse_mode="HTML")
            return ConversationHandler.END

        context.user_data["excel_path"] = file_path
        context.user_data["parser"] = parser
        context.user_data["file_name"] = document.file_name

        keyboard = ReplyKeyboardMarkup(ANALYSIS_TYPES, resize_keyboard=True)

        await processing_msg.edit_text(
            f"✅ <b>Fayl muvaffaqiyatli yuklandi!</b>\n\n"
            f"📄 Fayl: <code>{document.file_name}</code>\n"
            f"📊 Yozuvlar soni: {len(parser.df)}\n"
            f"💰 Debet jami: {parser.df['debet'].sum():,.2f} so'm\n"
            f"💳 Kredit jami: {parser.df['kredit'].sum():,.2f} so'm\n\n"
            f"Quyidagilardan birini tanlang:",
            reply_markup=keyboard,
            parse_mode="HTML",
        )

        return ASK_ANALYSIS_TYPE

    except Exception as e:
        await processing_msg.edit_text(
            f"❌ <b>Xatolik yuz berdi:</b>\n<code>{str(e)}</code>\n\nQaytadan urinib ko'ring."
        )
        return ConversationHandler.END


async def analyze_data(update: Update, context: ContextTypes.DEFAULT_TYPE):
    text = update.message.text
    parser = context.user_data.get("parser")
    file_name = context.user_data.get("file_name", "hisobot")

    if not parser:
        await update.message.reply_text(
            "❌ Ma'lumotlar topilmadi. Qaytadan boshlang.",
            reply_markup=ReplyKeyboardRemove(),
        )
        return ConversationHandler.END

    if text == "❌ Bekor qilish":
        await update.message.reply_text(
            "✅ Bekor qilindi.",
            reply_markup=ReplyKeyboardRemove(),
        )
        return ConversationHandler.END

    processing_msg = await update.message.reply_text(
        "⏳ <b>Tahlil qilinmoqda...</b>",
        parse_mode="HTML",
        reply_markup=ReplyKeyboardRemove(),
    )

    try:
        if text == "🔍 Xatolarni tekshirish":
            await _check_errors(processing_msg, parser)
        elif text == "📊 Balans tahlili":
            await _balance_analysis(processing_msg, parser)
        elif text == "📈 Daromadlar/xarajatlar":
            await _income_analysis(processing_msg, parser)
        elif text == "📑 To'liq hisobot":
            await _full_report(processing_msg, parser, update, context, file_name)
        else:
            await processing_msg.edit_text(
                "❌ Noto'g'ri tanlov. Iltimos, tugmalardan birini bosing.",
                reply_markup=ReplyKeyboardMarkup(ANALYSIS_TYPES, resize_keyboard=True),
            )
            return ASK_ANALYSIS_TYPE

        return ConversationHandler.END

    except Exception as e:
        await processing_msg.edit_text(
            f"❌ <b>Tahlilda xatolik:</b>\n<code>{str(e)}</code>\n\nQaytadan urinib ko'ring."
        )
        return ConversationHandler.END


async def _check_errors(msg, parser):
    errors = parser.detect_errors()

    if not errors:
        await msg.edit_text(
            "✅ <b>Xatolar topilmadi!</b>\n\n"
            "🎉 Barcha yozuvlar to'g'ri.\n"
            "📊 Debet va Kredit balansi teng.",
            parse_mode="HTML",
        )
    else:
        error_text = "⚠️ <b>Aniqlangan xatolar:</b>\n\n"
        critical_count = 0
        warning_count = 0

        for err in errors:
            if err["severity"] == "critical":
                emoji = "🔴"
                critical_count += 1
            else:
                emoji = "🟡"
                warning_count += 1

            error_text += f"{emoji} <b>{err['type']}:</b>\n{err['message']}\n\n"

        error_text += f"\n📊 Jami: {critical_count} ta jiddiy, {warning_count} ta ogohlantirish"

        await msg.edit_text(error_text, parse_mode="HTML")


async def _balance_analysis(msg, parser):
    analysis = parser.analyze_accounts()

    report_text = (
        f"📊 <b>Balans Tahlili</b>\n"
        f"{'='*30}\n\n"
        f"💰 Umumiy Debet: <code>{analysis['total_debet']:,.2f}</code> so'm\n"
        f"💳 Umumiy Kredit: <code>{analysis['total_kredit']:,.2f}</code> so'm\n"
        f"📈 Saldo: <code>{analysis['saldo']:,.2f}</code> so'm\n"
        f"📝 Hisob raqamlari: {analysis['accounts_count']} ta\n\n"
    )

    if analysis["by_account"]:
        report_text += "<b>Hisob raqamlari bo'yicha:</b>\n"
        for code, data in sorted(analysis["by_account"].items()):
            saldo = data['saldo']
            emoji = "🟢" if saldo >= 0 else "🔴"
            report_text += (
                f"{emoji} <code>{code}</code> {data['name']}\n"
                f"   Debet: {data['debet']:,.2f} | "
                f"Kredit: {data['kredit']:,.2f} | "
                f"Saldo: {saldo:,.2f}\n\n"
            )

    if abs(analysis['saldo']) < 0.01:
        report_text += "\n✅ <b>Balans to'g'ri!</b> Debet = Kredit"
    else:
        report_text += f"\n🔴 <b>Balans buzilgan!</b> Ayirma: {analysis['saldo']:,.2f}"

    await msg.edit_text(report_text, parse_mode="HTML")

        os.remove(report_path)

        pass

import pandas as pd
import openpyxl
from openpyxl.styles import Font, PatternFill, Alignment, Border, Side
from typing import Dict, List, Tuple
import logging

from config.settings import UZ_ACCOUNT_CODES

logger = logging.getLogger(__name__)


class ExcelParser:
    def __init__(self, file_path: str):
        self.file_path = file_path
        self.df = None
        self.workbook = None

    def load_file(self) -> bool:
        try:
            self.df = pd.read_excel(self.file_path)
            self.workbook = openpyxl.load_workbook(self.file_path)
            logger.info(f"Fayl yuklandi: {self.file_path}, {len(self.df)} ta qator")
            return True
        except Exception as e:
            logger.error(f"Faylni yuklashda xatolik: {e}")
            return False

    def validate_structure(self) -> Tuple[bool, List[str]]:
        errors = []

        if self.df is None:
            return False, ["Fayl yuklanmadi"]

        if len(self.df) == 0:
            return False, ["Fayl bo'sh! Ma'lumotlar yo'q."]

        self.df.columns = [str(col).strip().lower().replace(" ", "_") for col in self.df.columns]

        required_columns = ["hisob_raqami", "nomlanishi", "debet", "kredit"]
        missing = [col for col in required_columns if col not in self.df.columns]

        if missing:
            errors.append(f"Quyidagi ustunlar yetishmayapti: {', '.join(missing)}")
            errors.append("Majburiy ustunlar: hisob_raqami, nomlanishi, debet, kredit")
            return False, errors

        try:
            self.df["debet"] = pd.to_numeric(self.df["debet"], errors="coerce").fillna(0)
            self.df["kredit"] = pd.to_numeric(self.df["kredit"], errors="coerce").fillna(0)
        except Exception as e:
            errors.append(f"Raqamlarni konvertatsiya qilishda xatolik: {e}")

        empty_names = self.df[self.df["nomlanishi"].isna() | (self.df["nomlanishi"] == "")]
        if not empty_names.empty:
            errors.append(f"{len(empty_names)} ta yozuvda nomlanish kiritilmagan")

        invalid_codes = []
        for code in self.df["hisob_raqami"].astype(str):
            if not code.isdigit() or len(code) != 4:
                invalid_codes.append(code)

        if invalid_codes:
            errors.append(f"Noto'g'ri hisob raqamlari: {', '.join(set(invalid_codes))}")
            errors.append("Hisob raqamlari 4 xonali raqam bo'lishi kerak (masalan: 1010)")

        return len(errors) == 0, errors

    def analyze_accounts(self) -> Dict:
        if self.df is None:
            return {}

        analysis = {
            "total_debet": float(self.df["debet"].sum()),
            "total_kredit": float(self.df["kredit"].sum()),
            "saldo": float(self.df["debet"].sum() - self.df["kredit"].sum()),
            "accounts_count": len(self.df),
            "by_account": {},
        }

        if "hisob_raqami" in self.df.columns:
            grouped = self.df.groupby("hisob_raqami").agg({
                "debet": "sum",
                "kredit": "sum"
            }).reset_index()

            for _, row in grouped.iterrows():
                code = str(int(row["hisob_raqami"]))
                name = UZ_ACCOUNT_CODES.get(code, "Noma'lum hisob")

                analysis["by_account"][code] = {
                    "debet": float(row["debet"]),
                    "kredit": float(row["kredit"]),
                    "saldo": float(row["debet"] - row["kredit"]),
                    "name": name,
                }

        return analysis

    def detect_errors(self) -> List[Dict]:
        errors = []

        if self.df is None:
            return errors

        total_debet = self.df["debet"].sum()
        total_kredit = self.df["kredit"].sum()

        if abs(total_debet - total_kredit) > 0.01:
            errors.append({
                "type": "balans_xatosi",
                "message": (
                    f"Debet ({total_debet:,.2f}) va Kredit ({total_kredit:,.2f}) "
                    f"balansi buzilgan! Ayirma: {abs(total_debet - total_kredit):,.2f} so'm"
                ),
                "severity": "critical",
            })

        negative_debet = self.df[self.df["debet"] < 0]
        for _, row in negative_debet.iterrows():
            errors.append({
                "type": "manfiy_debet",
                "message": f"Manfiy debet: {row.get('nomlanishi', 'Noma\'lum')} - {row['debet']:,.2f} so'm",
                "severity": "warning",
            })

        negative_kredit = self.df[self.df["kredit"] < 0]
        for _, row in negative_kredit.iterrows():
            errors.append({
                "type": "manfiy_kredit",
                "message": f"Manfiy kredit: {row.get('nomlanishi', 'Noma\'lum')} - {row['kredit']:,.2f} so'm",
                "severity": "warning",
            })

        empty_names = self.df[self.df["nomlanishi"].isna() | (self.df["nomlanishi"] == "")]
        if not empty_names.empty:
            errors.append({
                "type": "bosh_nomlanish",
                "message": f"{len(empty_names)} ta yozuvda nomlanish kiritilmagan",
                "severity": "warning",
            })

        unknown_accounts = []
        for code in self.df["hisob_raqami"].astype(str):
            clean_code = str(int(float(code))) if code.replace('.', '').isdigit() else code
            if clean_code not in UZ_ACCOUNT_CODES:
                unknown_accounts.append(clean_code)

        if unknown_accounts:
            unique_unknown = list(set(unknown_accounts))[:5]
            errors.append({
                "type": "noma'lum_hisob",
                "message": (
                    f"Noma'lum hisob raqamlari: {', '.join(unique_unknown)}\n"
                    f"Jami {len(set(unknown_accounts))} ta noma'lum hisob"
                ),
                "severity": "warning",
            })

        zero_entries = self.df[(self.df["debet"] == 0) & (self.df["kredit"] == 0)]
        if not zero_entries.empty:
            errors.append({
                "type": "nol_yozuv",
                "message": f"{len(zero_entries)} ta yozuvda debet va kredit 0 ga teng",
                "severity": "warning",
            })

        return errors

from typing import Dict, List
from dataclasses import dataclass


@dataclass
class AccountEntry:
    code: str
    name: str
    debet: float
    kredit: float

    @property
    def saldo(self) -> float:
        return self.debet - self.kredit


class UzbekAccountingCalculator:
    """
    O'zbekiston buxgalteriya hisobi hisob-kitoblari
    BHMS 1, 3, 9, 15-sonli standartlar asosida
    """

    def __init__(self, entries: List[AccountEntry]):
        self.entries = entries

    def calculate_balance_sheet(self) -> Dict:
        assets = {
            "uzoq_muddatli_aktivlar": 0.0,
            "joriy_aktivlar": 0.0,
            "jami_aktivlar": 0.0,
        }

        liabilities = {
            "uzoq_muddatli_majburiyatlar": 0.0,
            "joriy_majburiyatlar": 0.0,
            "jami_majburiyatlar": 0.0,
        }

        equity = {
            "ustav_kapitali": 0.0,
            "qoshimcha_kapital": 0.0,
            "yillik_daromad": 0.0,
            "jami_kapital": 0.0,
        }

        for entry in self.entries:
            code = entry.code
            saldo = entry.saldo

            if code.startswith("1"):
                if code.startswith(("10", "11", "12", "13")):
                    assets["uzoq_muddatli_aktivlar"] += saldo
                else:
                    assets["joriy_aktivlar"] += saldo

            elif code.startswith(("2", "3")):
                if code.startswith(("30", "31")):
                    liabilities["uzoq_muddatli_majburiyatlar"] += entry.kredit
                elif code.startswith(("32", "33", "34")):
                    liabilities["joriy_majburiyatlar"] += entry.kredit

            elif code.startswith("4"):
                if code.startswith("41"):
                    equity["ustav_kapitali"] += entry.kredit
                elif code.startswith("42"):
                    equity["qoshimcha_kapital"] += entry.kredit
                elif code.startswith("43"):
                    equity["yillik_daromad"] += saldo

        assets["jami_aktivlar"] = (
            assets["uzoq_muddatli_aktivlar"] + assets["joriy_aktivlar"]
        )
        liabilities["jami_majburiyatlar"] = (
            liabilities["uzoq_muddatli_majburiyatlar"] + 
            liabilities["joriy_majburiyatlar"]
        )
        equity["jami_kapital"] = (
            equity["ustav_kapitali"] + 
            equity["qoshimcha_kapital"] + 
            equity["yillik_daromad"]
        )

        balance_check = abs(
            assets["jami_aktivlar"] - 
            (liabilities["jami_majburiyatlar"] + equity["jami_kapital"])
        )

        return {
            "aktivlar": assets,
            "majburiyatlar": liabilities,
            "kapital": equity,
            "balans_tekshiruvi": balance_check < 0.01,
            "balans_ayirma": balance_check,
        }

    def calculate_income_statement(self) -> Dict:
        revenue = 0.0
        cost_of_sales = 0.0
        operating_expenses = 0.0
        financial_expenses = 0.0
        other_income = 0.0

        for entry in self.entries:
            code = entry.code

            if code.startswith("5"):
                revenue += entry.kredit
            elif code.startswith("6"):
                cost_of_sales += entry.debet
            elif code.startswith("7"):
                operating_expenses += entry.debet
            elif code.startswith("8"):
                operating_expenses += entry.debet
            elif code.startswith("9"):
                financial_expenses += entry.debet
            elif code.startswith("10"):
                other_income += entry.kredit

        gross_profit = revenue - cost_of_sales
        operating_profit = gross_profit - operating_expenses
        profit_before_tax = operating_profit - financial_expenses + other_income

        tax_rate = 0.15
        income_tax = profit_before_tax * tax_rate if profit_before_tax > 0 else 0
        net_profit = profit_before_tax - income_tax

        return {
            "realizatsiya_daromadi": revenue,
            "realizatsiya_tannarxi": cost_of_sales,
            "vali_foyda": gross_profit,
            "boshqaruv_xarajatlari": operating_expenses,
            "sotish_xarajatlari": 0.0,
            "savdo_foydasi": operating_profit,
            "moliyaviy_xarajatlar": financial_expenses,
            "boshqa_daromadlar": other_income,
            "solig_oldi_foyda": profit_before_tax,
            "foyda_solig'i": income_tax,
            "sof_foyda": net_profit,
        }

    def calculate_cash_flow(self) -> Dict:
        operating = 0.0
        investing = 0.0
        financing = 0.0

        for entry in self.entries:
            code = entry.code

            if code.startswith(("50", "51", "52")):
                operating += entry.debet - entry.kredit
            elif code.startswith(("08", "58")):
                investing += entry.debet - entry.kredit
            elif code.startswith(("66", "67")):
                financing += entry.kredit - entry.debet

        net_cash = operating + investing + financing

        return {
            "operatsion_faoliyat": operating,
            "investitsion_faoliyat": investing,
            "moliyaviy_faoliyat": financing,
            "naqd_pul_oshishi": net_cash,
        }

import re
from typing import Tuple, List


def validate_account_code(code: str) -> Tuple[bool, str]:
    """Hisob raqamini tekshirish"""
    if not code:
        return False, "Hisob raqami bo'sh"

    if not re.match(r'^\d{4}$', code):
        return False, "Hisob raqami 4 xonali raqam bo'lishi kerak"

    return True, "To'g'ri"


def validate_amount(amount) -> Tuple[bool, str]:
    """Summani tekshirish"""
    try:
        value = float(amount)
        if value < 0:
            return False, "Summa manfiy bo'lishi mumkin emas"
        return True, "To'g'ri"
    except (ValueError, TypeError):
        return False, "Noto'g'ri summa formati"


def validate_excel_data(data: List[dict]) -> Tuple[bool, List[str]]:
    """Excel ma'lumotlarini tekshirish"""
    errors = []

    if not data:
        return False, ["Ma'lumotlar bo'sh"]

    required_fields = ["hisob_raqami", "nomlanishi", "debet", "kredit"]

    for i, row in enumerate(data, 1):
        for field in required_fields:
            if field not in row:
                errors.append(f"Qator {i}: '{field}' ustuni yetishmayapti")

        if "hisob_raqami" in row:
            valid, msg = validate_account_code(str(row["hisob_raqami"]))
            if not valid:
                errors.append(f"Qator {i}: {msg}")

        if "debet" in row:
            valid, msg = validate_amount(row["debet"])
            if not valid:
                errors.append(f"Qator {i}: Debet - {msg}")

        if "kredit" in row:
            valid, msg = validate_amount(row["kredit"])
            if not valid:
                errors.append(f"Qator {i}: Kredit - {msg}")

    return len(errors) == 0, errors

from datetime import datetime


def format_amount(amount: float, currency: str = "so'm") -> str:
    """Summani formatlash"""
    return f"{amount:,.2f} {currency}"


def format_date(date: datetime = None) -> str:
    """Sanani formatlash"""
    if date is None:
        date = datetime.now()
    return date.strftime("%d.%m.%Y")


def format_percentage(value: float, decimals: int = 2) -> str:
    """Foizni formatlash"""
    return f"{value:.{decimals}f}%"


def escape_html(text: str) -> str:
    """HTML maxsus belgilarini qochish"""
    return text.replace("&", "&amp;").replace("<", "&lt;").replace(">", "&gt;")
