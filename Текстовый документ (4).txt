# Helper functions
def set_cell_shading(cell, color):
    shading = OxmlElement('w:shd')
    shading.set(qn('w:fill'), color)
    cell._tc.get_or_add_tcPr().append(shading)

def set_run_font(run, font_name='Calibri', size=11, bold=False, color=None):
    run.font.name = font_name
    run.font.size = Pt(size)
    run.font.bold = bold
    if color:
        run.font.color.rgb = RGBColor(*color)

def add_heading_custom(doc, text, level=1):
    heading = doc.add_heading(text, level=level)
    for run in heading.runs:
        run.font.name = 'Calibri'
        run.font.color.rgb = RGBColor(0, 51, 102) if level == 1 else RGBColor(0, 102, 153)
        run.font.bold = True
    return heading

engine_code = '''from config import Config
from decimal import Decimal, ROUND_HALF_UP

class UzbekTaxCalculator:
    """O'zbekiston soliq va buxgalteriya hisoblash dvigateli"""
    
    def __init__(self):
        self.config = Config()
    
    def calculate_salary(self, gross_salary: float, tax_regime: str = 'general') -> dict:
        """
        Ish haqi hisoblash
        
        Args:
            gross_salary: Yalpi ish haqi (so'm)
            tax_regime: Soliq rejimi
            
        Returns:
            dict: Barcha ushlanmalar va sof ish haqi
        """
        gross = Decimal(str(gross_salary))
        tax_free = Decimal(str(self.config.TAX_FREE_MINIMUM))
        
        # Soliq solinadigan qism
        taxable_amount = max(gross - tax_free, Decimal('0'))
        
        # JSHDS (12%)
        jshds = (taxable_amount * Decimal(str(self.config.PIT_RATE))).quantize(Decimal('0.01'), rounding=ROUND_HALF_UP)
        
        # Ijtimoiy sug'urta (4%)
        social_insurance = (gross * Decimal(str(self.config.SOCIAL_INSURANCE_EMP))).quantize(Decimal('0.01'))
        
        # Ishsizlik sug'urtasi (0.5%)
        unemployment = (gross * Decimal(str(self.config.UNEMPLOYMENT_EMP))).quantize(Decimal('0.01'))
        
        # Uy-joy jamg'armasi (1%)
        housing = (gross * Decimal(str(self.config.HOUSING_EMP))).quantize(Decimal('0.01'))
        
        # INCES kasbiy ta'lim (0.5%)
        inces = (gross * Decimal(str(self.config.INCES_EMP))).quantize(Decimal('0.01'))
        
        # Jami ushlanmalar
        total_deductions = jshds + social_insurance + unemployment + housing + inces
        
        # Sof ish haqi
        net_salary = gross - total_deductions
        
        return {
            'gross_salary': float(gross),
            'tax_free_amount': float(tax_free),
            'taxable_amount': float(taxable_amount),
            'jshds': float(jshds),
            'social_insurance': float(social_insurance),
            'unemployment': float(unemployment),
            'housing': float(housing),
            'inces': float(inces),
            'total_deductions': float(total_deductions),
            'net_salary': float(net_salary),
            'deduction_rate': float(total_deductions / gross * 100) if gross > 0 else 0
        }
    
    def calculate_employer_costs(self, gross_salary: float) -> dict:
        """Ish beruvchi to'lovlari"""
        gross = Decimal(str(gross_salary))
        
        # Ijtimoiy soliq (12%)
        social_tax = (gross * Decimal(str(self.config.SOCIAL_TAX_EMPLOYER))).quantize(Decimal('0.01'))
        
        # Pensiya jamg'armasi (0.1%)
        pension_fund = (gross * Decimal(str(self.config.PENSION_FUND_EMPLOYER))).quantize(Decimal('0.01'))
        
        total = gross + social_tax + pension_fund
        
        return {
            'gross_salary': float(gross),
            'social_tax': float(social_tax),
            'pension_fund': float(pension_fund),
            'total_cost': float(total),
            'employer_rate': float((social_tax + pension_fund) / gross * 100) if gross > 0 else 0
        }
    
    def calculate_vat(self, amount: float, include_vat: bool = True) -> dict:
        """QQS hisoblash"""
        if include_vat:
            # QQSni ajratish (12%)
            vat_amount = (Decimal(str(amount)) * Decimal('0.12') / Decimal('1.12')).quantize(Decimal('0.01'))
            base_amount = Decimal(str(amount)) - vat_amount
        else:
            # QQS qo'shish
            base_amount = Decimal(str(amount))
            vat_amount = (base_amount * Decimal('0.12')).quantize(Decimal('0.01'))
            
        return {
            'base_amount': float(base_amount),
            'vat_amount': float(vat_amount),
            'total_amount': float(base_amount + vat_amount),
            'vat_rate': 12
        }
    
    def calculate_turnover_tax(self, turnover: float, regime: str = 'standard') -> dict:
        """Aylanma solig'i hisoblash"""
        t = Decimal(str(turnover))
        
        if regime == 'preferential':
            rate = Decimal(str(self.config.TURNOVER_TAX_PREFERENTIAL))
        elif regime == 'simplified_vat':
            rate = Decimal(str(self.config.SIMPLIFIED_VAT))
        else:
            rate = Decimal(str(self.config.TURNOVER_TAX_STANDARD))
            
        tax = (t * rate).quantize(Decimal('0.01'))
        
        return {
            'turnover': float(t),
            'tax_amount': float(tax),
            'rate': float(rate * 100),
            'regime': regime
        }
    
    def calculate_profit_tax(self, profit: float, is_bank: bool = False) -> dict:
        """Foyda solig'i hisoblash"""
        p = Decimal(str(profit))
        rate = Decimal(str(self.config.PROFIT_TAX_BANK if is_bank else self.config.PROFIT_TAX_RATE))
        tax = (p * rate).quantize(Decimal('0.01'))
        
        return {
            'profit': float(p),
            'tax_amount': float(tax),
            'rate': float(rate * 100),
            'net_profit': float(p - tax)
        }

class UzbekTaxCalculator:
    \"\"\"O'zbekiston soliq va buxgalteriya hisoblash dvigateli\"\"\"
    
    def __init__(self):
        self.config = Config()
    
    def calculate_salary(self, gross_salary: float, tax_regime: str = 'general') -> dict:
        \"\"\"
        Ish haqi hisoblash
        
        Args:
            gross_salary: Yalpi ish haqi (so'm)
            tax_regime: Soliq rejimi
            
        Returns:
            dict: Barcha ushlanmalar va sof ish haqi
        \"\"\"
        gross = Decimal(str(gross_salary))
        tax_free = Decimal(str(self.config.TAX_FREE_MINIMUM))
        
        # Soliq solinadigan qism
        taxable_amount = max(gross - tax_free, Decimal('0'))
        
        # JSHDS (12%)
        jshds = (taxable_amount * Decimal(str(self.config.PIT_RATE))).quantize(Decimal('0.01'), rounding=ROUND_HALF_UP)
        
        # Ijtimoiy sug'urta (4%)
        social_insurance = (gross * Decimal(str(self.config.SOCIAL_INSURANCE_EMP))).quantize(Decimal('0.01'))
        
        # Ishsizlik sug'urtasi (0.5%)
        unemployment = (gross * Decimal(str(self.config.UNEMPLOYMENT_EMP))).quantize(Decimal('0.01'))
        
        # Uy-joy jamg'armasi (1%)
        housing = (gross * Decimal(str(self.config.HOUSING_EMP))).quantize(Decimal('0.01'))
        
        # INCES kasbiy ta'lim (0.5%)
        inces = (gross * Decimal(str(self.config.INCES_EMP))).quantize(Decimal('0.01'))
        
        # Jami ushlanmalar
        total_deductions = jshds + social_insurance + unemployment + housing + inces
        
        # Sof ish haqi
        net_salary = gross - total_deductions
        
        return {
            'gross_salary': float(gross),
            'tax_free_amount': float(tax_free),
            'taxable_amount': float(taxable_amount),
            'jshds': float(jshds),
            'social_insurance': float(social_insurance),
            'unemployment': float(unemployment),
            'housing': float(housing),
            'inces': float(inces),
            'total_deductions': float(total_deductions),
            'net_salary': float(net_salary),
            'deduction_rate': float(total_deductions / gross * 100) if gross > 0 else 0
        }

        \"\"\"Ish beruvchi to'lovlari\"\"\"

        return {
            'gross_salary': float(gross),
            'social_tax': float(social_tax),
            'pension_fund': float(pension_fund),
            'total_cost': float(total),

        }

        \"\"\"QQS hisoblash\"\"\"

        }

# test_accounting.py
import unittest
from accounting_engine import UzbekTaxCalculator

class TestSalaryCalculation(unittest.TestCase):
    def setUp(self):
        self.calc = UzbekTaxCalculator()
    
    def test_salary_with_tax_free(self):
        # Soliqdan ozod minimumdan yuqori ish haqi
        result = self.calc.calculate_salary(3_000_000)
        self.assertEqual(result['tax_free_amount'], 1_650_000)
        self.assertAlmostEqual(result['taxable_amount'], 1_350_000, places=2)
        self.assertAlmostEqual(result['jshds'], 162_000, places=2)  # 1_350_000 * 0.12
    
    def test_salary_below_tax_free(self):
        # Soliqdan ozod minimumdan past ish haqi
        result = self.calc.calculate_salary(1_000_000)
        self.assertEqual(result['taxable_amount'], 0)
        self.assertEqual(result['jshds'], 0)
    
    def test_total_deductions(self):
        result = self.calc.calculate_salary(5_000_000)
        expected_deductions = (
            result['jshds'] + 
            result['social_insurance'] + 
            result['unemployment'] + 
            result['housing'] + 
            result['inces']
        )
        self.assertAlmostEqual(result['total_deductions'], expected_deductions, places=2)
    
    def test_employer_costs(self):
        result = self.calc.calculate_employer_costs(3_000_000)
        self.assertAlmostEqual(result['social_tax'], 360_000, places=2)  # 3_000_000 * 0.12
        self.assertAlmostEqual(result['pension_fund'], 3_000, places=2)  # 3_000_000 * 0.001

class TestVATCalculation(unittest.TestCase):
    def setUp(self):
        self.calc = UzbekTaxCalculator()
    
    def test_vat_extraction(self):
        result = self.calc.calculate_vat(1_120_000, include_vat=True)
        self.assertAlmostEqual(result['vat_amount'], 120_000, places=2)
        self.assertAlmostEqual(result['base_amount'], 1_000_000, places=2)
    
    def test_vat_addition(self):
        result = self.calc.calculate_vat(1_000_000, include_vat=False)
        self.assertAlmostEqual(result['vat_amount'], 120_000, places=2)
        self.assertAlmostEqual(result['total_amount'], 1_120_000, places=2)
